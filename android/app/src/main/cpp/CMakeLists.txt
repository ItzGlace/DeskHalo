
cmake_minimum_required(VERSION 3.22.1)
project(deskhalo)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
set(ASSIMP_WARNINGS_AS_ERRORS OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ASSIMP_TOOLS OFF CACHE BOOL "" FORCE)
set(ASSIMP_INSTALL OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ALL_IMPORTERS_BY_DEFAULT OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_OBJ_IMPORTER ON CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_GLTF_IMPORTER ON CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_FBX_IMPORTER ON CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ALL_EXPORTERS_BY_DEFAULT OFF CACHE BOOL "" FORCE)
set(ASSIMP_NO_EXPORT ON CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ZLIB ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
FetchContent_Declare(assimp URL https://github.com/assimp/assimp/archive/refs/tags/v5.4.3.tar.gz)

FetchContent_MakeAvailable(assimp)
# Ship upstream and bundled dependency notices with the APK.
set(MODEL_NOTICE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../../build/generated/model-assets/licenses/assimp")
file(MAKE_DIRECTORY "${MODEL_NOTICE_DIR}")
configure_file("${assimp_SOURCE_DIR}/LICENSE" "${MODEL_NOTICE_DIR}/LICENSE" COPYONLY)
file(GLOB_RECURSE MODEL_LICENSES
    "${assimp_SOURCE_DIR}/contrib/*LICENSE*" "${assimp_SOURCE_DIR}/contrib/*license*"
    "${assimp_SOURCE_DIR}/contrib/*COPYING*")
foreach(NOTICE IN LISTS MODEL_LICENSES)
    if(NOT IS_DIRECTORY "${NOTICE}")
        file(RELATIVE_PATH RELATIVE_NOTICE "${assimp_SOURCE_DIR}" "${NOTICE}")
        get_filename_component(NOTICE_PARENT "${MODEL_NOTICE_DIR}/${RELATIVE_NOTICE}" DIRECTORY)
        file(MAKE_DIRECTORY "${NOTICE_PARENT}")
        configure_file("${NOTICE}" "${MODEL_NOTICE_DIR}/${RELATIVE_NOTICE}" COPYONLY)
    endif()
endforeach()

find_package(OpenXR REQUIRED CONFIG)
add_library(deskhalo SHARED native-lib.cpp)
# Assimp sets a global debug postfix; JNI requires this exact name in every build.
set_target_properties(deskhalo PROPERTIES DEBUG_POSTFIX "" OUTPUT_NAME "deskhalo")
target_compile_definitions(deskhalo PRIVATE XR_USE_PLATFORM_ANDROID XR_USE_GRAPHICS_API_OPENGL_ES)
target_link_libraries(deskhalo assimp OpenXR::openxr_loader android log EGL GLESv3 jnigraphics)
